% Records are passed and returned by value.
%? stdout: 4.00 6.00
%? stdout: 1.00 2.00
%? stdout: 3.00 4.00
%? stdout: 2
_add input parameter list [record #pt d2, record #pt d3]
output parameter list [record #pt d4, int b2];
	d4 <--- d2 + d3;
	b2 <--- 2;
	return [d4, b2];
end

_main
	record #pt
		type real : x;
		type real : y;
	endrecord
	type #pt : d2;
	type #pt : d3;
	type #pt : d4;
	type int : b2;
	d2.x <--- 1.00;
	d2.y <--- 2.00;
	d3.x <--- 3.00;
	d3.y <--- 4.00;
	[d4, b2] <--- call _add with parameters [d2, d3];
	write(d4);
	write(d2);
	write(d3);
	write(b2);
	return;
end
