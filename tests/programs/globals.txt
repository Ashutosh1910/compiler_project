% Globals are shared by every function, including functions defined before
% the declaration.
%? stdout: 5
%? stdout: 15
%? stdout: 2.50 3.50
%? stdout: 16
_bump input parameter list [int b2]
output parameter list [int b3];
	b6 <--- b6 + b2;
	d6.x <--- d6.x + 1.00;
	d6.y <--- d6.y + 1.00;
	b3 <--- b6;
	return [b3];
end

_main
	record #pt
		type real : x;
		type real : y;
	endrecord
	type int : b6 : global;
	type #pt : d6 : global;
	type int : b2;
	type int : b3;
	b6 <--- 5;
	write(b6);
	d6.x <--- 1.50;
	d6.y <--- 2.50;
	b2 <--- 10;
	[b3] <--- call _bump with parameters [b2];
	write(b6);
	write(d6);
	b2 <--- 1;
	[b6] <--- call _bump with parameters [b2];
	write(b6);
	return;
end
