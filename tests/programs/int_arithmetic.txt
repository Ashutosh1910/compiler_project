% Integer arithmetic: precedence, left associativity, division truncating
% towards zero.
%? stdout: 10
%? stdout: 4
%? stdout: 21
%? stdout: 2
%? stdout: -2
%? stdout: 3
%? stdout: 6
%? stdout: 13
%? stdout: 20
%? stdout: 2
%? stdout: -5
%? stdout: 1
%? stdout: -1
%? stdout: 0
_main
	type int : b2;
	type int : b3;
	type int : b4;
	b2 <--- 7;
	b3 <--- 3;
	b4 <--- b2 + b3;
	write(b4);
	b4 <--- b2 - b3;
	write(b4);
	b4 <--- b2 * b3;
	write(b4);
	b4 <--- b2 / b3;
	write(b4);
	b4 <--- (0 - b2) / b3;
	write(b4);
	b4 <--- b2 - b3 - 1;
	write(b4);
	b4 <--- b2 / b3 * b3;
	write(b4);
	b4 <--- b2 + b3 * 2;
	write(b4);
	b4 <--- (b2 + b3) * 2;
	write(b4);
	b4 <--- 100 / 10 / 5;
	write(b4);
	b4 <--- 0 - 5;
	write(b4);
	b4 <--- (0 - 7) - (0 - 8);
	write(b4);
	b4 <--- 7 / (0 - 7);
	write(b4);
	b4 <--- 3 / 7;
	write(b4);
	return;
end
