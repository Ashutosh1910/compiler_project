% Frames of every size still call printf with an aligned stack (printf
% crashes on a misaligned stack when printing reals).
% main writes 1; _three(2) writes c5 = 5; _two(4, 1) writes 5; _one(5)
% writes 5; _two returns 5 + 1 = 6; _three writes 6 and returns
% 6 + 4 + 5 + 2 = 17
%? stdout: 1.00
%? stdout: 5.00
%? stdout: 5.00
%? stdout: 5.00
%? stdout: 6.00
%? stdout: 17.00
_one input parameter list [real c2]
output parameter list [real c3];
	write(c2);
	c3 <--- c2;
	return [c3];
end

_two input parameter list [real c2, int b2]
output parameter list [real c3];
	type real : c4;
	c4 <--- c2 + 1.00;
	write(c4);
	[c3] <--- call _one with parameters [c4];
	c3 <--- c3 + b2;
	return [c3];
end

_three input parameter list [real c2]
output parameter list [real c3, int b3];
	type real : c4;
	type real : c5;
	type int : b4;
	c4 <--- c2 + 2.00;
	c5 <--- c4 + 1.00;
	b4 <--- 1;
	write(c5);
	[c3] <--- call _two with parameters [c4, b4];
	write(c3);
	c3 <--- c3 + c4 + c5 + c2;
	b3 <--- 0;
	return [c3, b3];
end

_main
	type real : c2;
	type real : c3;
	type int : b2;
	c2 <--- 1.00;
	write(c2);
	c2 <--- 2.00;
	[c3, b2] <--- call _three with parameters [c2];
	write(c3);
	return;
end
