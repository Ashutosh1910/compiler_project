% Elements of a real array: int values are converted on store; reals print
% with two decimals.
%? stdout: 3.00
%? stdout: 1.50
%? stdout: 4.50|-0.25
%? stdout: 1
_main
	type real[3] : c2;
	type int : b2;
	c2[1] <--- 3;
	b2 <--- 2;
	c2[b2] <--- c2[1] / b2;
	write(c2[1]);
	write(c2[2]);
	c2[0] <--- 0.25 - c2[0] - 0.50;
	print(c2[1] + c2[2], "|", c2[0], "\n");
	if (c2[2] < 2) then
		write(1);
	else
		write(0);
	endif
	return;
end
