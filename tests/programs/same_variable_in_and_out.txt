% A variable may be both an argument and a result of the same call.
%? stdout: 2
%? stdout: 4
%? stdout: 3 5
_double input parameter list [int b2]
output parameter list [int b3];
	b3 <--- b2 * 2;
	return [b3];
end

_swap input parameter list [record #p b2]
output parameter list [record #p b3];
	b3.a <--- b2.b;
	b3.b <--- b2.a;
	return [b3];
end

_main
	record #p
		type int : a;
		type int : b;
	endrecord
	type int : b2;
	type #p : d2;
	b2 <--- 1;
	[b2] <--- call _double with parameters [b2];
	write(b2);
	[b2] <--- call _double with parameters [b2];
	write(b2);
	d2.a <--- 5;
	d2.b <--- 3;
	[d2] <--- call _swap with parameters [d2];
	write(d2);
	return;
end
