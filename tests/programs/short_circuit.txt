% &&& and @@@ evaluate their right operand only when needed, so a guard can
% protect an array access.
%? stdout: 3
%? stdout: 1
_main
	type int[3] : c2;
	type int : b2;
	b2 <--- 0;
	while ((b2 < 3) &&& (c2[b2] == 0))
		b2 <--- b2 + 1;
	endwhile
	write(b2);
	if ((b2 >= 3) @@@ (c2[b2] == 1)) then
		write(1);
	else
		write(0);
	endif
	return;
end
