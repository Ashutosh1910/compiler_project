% A return list is a parallel assignment to the output parameters: a
% returned output parameter contributes its value from before the return.
%? stdout: 7
%? stdout: 70
%? stdout: 2
%? stdout: 1
%? stdout: 5 6
%? stdout: 3 4
_pair input parameter list [int b2]
output parameter list [int b3, int b4];
	b3 <--- b2 * 10;
	return [b2, b3];
end

_swap input parameter list [int b2]
output parameter list [int b3, int b4];
	b3 <--- 1;
	b4 <--- 2;
	return [b4, b3];
end

_rotate input parameter list [record #p d2]
output parameter list [record #p d3, record #p d4];
	d3 <--- d2;
	d4.a <--- d2.a + 2;
	d4.b <--- d2.b + 2;
	return [d4, d3];
end

_main
	record #p
		type int : a;
		type int : b;
	endrecord
	type int : b2;
	type int : b5;
	type int : b6;
	type #p : d2;
	type #p : d3;
	type #p : d4;
	b2 <--- 7;
	[b5, b6] <--- call _pair with parameters [b2];
	write(b5);
	write(b6);
	[b5, b6] <--- call _swap with parameters [b2];
	write(b5);
	write(b6);
	d2.a <--- 3;
	d2.b <--- 4;
	[d3, d4] <--- call _rotate with parameters [d2];
	write(d3);
	write(d4);
	return;
end
